// Edge cases for the app's DOCX writer on synthetic data: (1) Unicode beyond BMP, combining marks, XML specials, CRLF;
// (2) a C0 control character typed into a header field; (3) a bundle with no rows.
import { writeFileSync } from 'node:fs';
const W = process.argv[2], OUT = process.argv[3];
const { indexDocx } = await import(`${W}/src/lib/views/bundles/docx.ts`);
const { docxText } = await import(`${W}/scripts/parity/lib/exports.mjs`);
const H = { court: 'Synthetic Court', judge: 'Division', intro: 'In the matter between:', caseNo: 'X/1', claimant: 'Alpha & <Beta> "Ltd"\r\nline two', claimantRole: 'PLAINTIFF', defendant: 'Gamma', defendantRole: 'DEFENDANT' };
const row = (n: number, desc: string) => ({ n, key: 'K' + n, desc, date: '', blank: false, missing: false, review: false, pages: 2, from: n * 2 - 1, to: n * 2, sliceUrl: '', caselines: null, parts: [] });
const B = (rows: unknown[]) => ({ no: 'A', name: 'Synthetic', title: 'Synthetic bundle', note: '', rows, pages: 4, indexPages: 1, file: '', indexFile: '' });
const cases: Record<string, Uint8Array> = {
	unicode: indexDocx([B([row(1, 'Café \u{1F600} é ünïcödé'), row(2, 'a & b < c > d "q"')])], H, 'Creator'),
	control: indexDocx([B([row(1, 'ok')])], { ...H, court: 'Synthetic\u000bCourt' }, 'Creator'),
	norows: indexDocx([B([])], H, 'Creator'),
};
for (const [k, v] of Object.entries(cases)) { writeFileSync(`${OUT}/${k}.docx`, v); let t; try { t = docxText(Buffer.from(v)).split('\n').length + ' paragraphs'; } catch (e) { t = 'docxText error ' + e; } console.log(k, v.length, 'bytes;', t); }
